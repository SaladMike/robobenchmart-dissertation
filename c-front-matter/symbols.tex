\chapter*{Symbols}
\addcontentsline{toc}{chapter}{Symbols}

\noindent
\begin{tabular}{>{$}p{2.5cm}<{$}p{10cm}}
\mathbf{p}_i & Vertex $i$ of a polygonal boundary \\
\mathbf{u}_i & Directed edge from $\mathbf{p}_i$ to $\mathbf{p}_{i+1}$ \\
\ell_i & Length of boundary edge $i$ \\
\theta_i & Orientation of boundary edge $i$ \\
\mathbf{T}_i & Basis tensor induced by boundary sample $i$ \\
\mathbf{T}(\mathbf{x}) & Aggregated tensor field at floor point $\mathbf{x}$ \\
\theta^{\star}(\mathbf{x}) & Preferred undirected orientation at $\mathbf{x}$ \\
d & Spatial decay coefficient in tensor-field aggregation \\
D & Maximum boundary-segment length after subdivision \\
P & Number of vertices in a boundary polygon \\
\mathbf{a}_t & 13-dimensional action vector at step $t$ \\
y_i & Success flag of episode $i$ \\
k & Number of successful episodes \\
n & Number of episodes \\
\hat{p} & Observed success rate, $k/n$ \\
z & Standard normal quantile in the Wilson interval ($1.96$ for 95\%) \\
\end{tabular}
